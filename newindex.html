<script>

    // ⚠️ REPLACE THIS WITH YOUR DEPLOYED GOOGLE APPS SCRIPT WEB APP URL

    const DEPLOYED_APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbylTZJeMPsqvDQd4nCZCo48w5mz5k7jXx20tPQSEdHYnl2PrimFIGJ0vNDOupkG8XB0/exec";


    let globalSessionData = null;


    function showFeedback(text, isError = false) {

        const msgBox = document.getElementById('feedbackMessage');

        msgBox.textContent = text;

        msgBox.className = `alert ${isError ? 'alert-error' : 'alert-success'}`;

        msgBox.classList.remove('hidden');

    }


    function clearFeedback() {

        document.getElementById('feedbackMessage').classList.add('hidden');

    }


    async function handlePortalLogin() {

        const sName = document.getElementById('studentName').value.trim();

        const invNo = document.getElementById('invoiceNumber').value.trim();

        const loginBtn = document.getElementById('loginBtn');


        if (!sName || !invNo) {

            showFeedback("Please complete all profile authentication parameters.", true);

            return;

        }


        clearFeedback();

        loginBtn.disabled = true;

        loginBtn.innerHTML = '<div class="loader"></div> Verifying Identity Matrix...';


        try {

            const url = `${DEPLOYED_APPS_SCRIPT_URL}?action=fetchDetails&studentName=${encodeURIComponent(sName)}&invoiceNumber=${encodeURIComponent(invNo)}`;

            

            const response = await fetch(url);

            const textResult = await response.text();

            

            // Check for raw processing errors handled by our backend engine

            if (textResult.startsWith("ERROR") || textResult.startsWith("INVALID")) {

                throw new Error(textResult);

            }


            const data = JSON.parse(textResult);

            

            if (data.status === "success") {

                globalSessionData = data;

                

                // Populate Dashboard Metrics

                document.getElementById('welcomeHeader').textContent = `✅ ${data.studentName}`;

                document.getElementById('metaClass').textContent = data.class;

                document.getElementById('metaFees').textContent = data.totalFees;

                document.getElementById('metaPaid').textContent = data.amountPaid;

                document.getElementById('metaBalance').textContent = `$${data.remainingBalance}`;

                

                // Swap Interface Panels

                document.getElementById('loginWindow').classList.add('hidden');

                document.getElementById('dashboardWindow').classList.remove('hidden');

            }

        } catch (err) {

            showFeedback(`Authentication Mismatch: ${err.message}`, true);

        } finally {

            loginBtn.disabled = false;

            loginBtn.textContent = 'Access Statement Dashboard';

        }

    }


    function handleReceiptUpload() {

        const fileInput = document.getElementById('receiptFile');

        const uploadBtn = document.getElementById('uploadBtn');

        const files = fileInput.files;


        if (!files || files.length === 0) {

            showFeedback("Please attach a valid file capture payload first.", true);

            return;

        }


        clearFeedback();

        uploadBtn.disabled = true;

        uploadBtn.innerHTML = '<div class="loader"></div> Injecting Slip into Google Drive...';


        const file = files[0];

        const reader = new FileReader();

        reader.onload = async function(e) {

            const base64Data = e.target.result;


            try {

                // Post transmission payloads securely into the matrix boundary

                const formData = new URLSearchParams();

                formData.append('action', 'uploadReceipt');

                formData.append('studentName', globalSessionData.studentName);

                formData.append('invoiceNumber', globalSessionData.invoiceNumber);

                formData.append('fileData', base64Data);


                const response = await fetch(DEPLOYED_APPS_SCRIPT_URL, {

                    method: 'POST',

                    body: formData,

                    headers: { 'Content-Type': 'application/x-www-form-urlencoded' }

                });


                const resData = await response.json();

                

                if (resData.status === "success") {

                    showFeedback("🎉 Transmission Verified! Receipt successfully logged to Drive archiving.");

                    fileInput.value = ""; // Clear file selector input

                } else {

                    throw new Error("Internal processing server exception.");

                }

            } catch (err) {

                showFeedback(`Upload Routine Interrupted: ${err.message}`, true);

            } finally {

                uploadBtn.disabled = false;

                uploadBtn.textContent = 'Submit Verification Slip';

            }

        };

        

        reader.readAsDataURL(file);

    }
